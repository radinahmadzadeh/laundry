{# Home script: courier request for a ready order. #} function submitCourier(orderId){ var latEl = document.getElementById('lat-' + orderId); var postalEl = document.getElementById('postal-' + orderId); var lat = latEl.value; var lng = document.getElementById('lng-' + orderId).value; var postal = normalizePostal(postalEl.value); if (!lat){ flashInvalid(document.getElementById('interactive-map-' + orderId), 'لطفاً موقعیت خود را روی نقشه انتخاب کنید.'); return; } if (!/^\d{10}$/.test(postal)){ flashInvalid(postalEl, 'کد پستی باید ۱۰ رقم باشد.'); return; } // Not confirmed for this address yet: the form-error toast explains and points at it. if (!postalEl.checkValidity()) return; var btn = document.getElementById('courier-submit-' + orderId); if (!btn || btn.disabled) return; var originalText = btn.innerHTML; btn.innerHTML = ' در حال ثبت درخواست...'; btn.disabled = true; btn.classList.add('opacity-75', 'cursor-not-allowed'); fetch('/request-courier/', { method: 'POST', headers: { 'Content-Type': 'application/json', 'X-CSRFToken': getOrderCsrfToken() }, body: JSON.stringify({ order_id: orderId, lat: lat, lng: lng, postal: postal, address: getCourierAddress(orderId) }) }) .then(function(response){ return response.json(); }) .then(function(data){ if (data.status === 'success'){ // Reload this order so the page shows the saved address and live courier status. showToast(data.message || 'درخواست پیک ثبت شد.'); setTimeout(function(){ window.location.href = '{% url "home" %}?order_id=' + encodeURIComponent(orderId) + '#track-section'; }, 900); } else { btn.innerHTML = originalText; btn.disabled = false; btn.classList.remove('opacity-75', 'cursor-not-allowed'); showToast(data.message || 'ثبت درخواست پیک انجام نشد؛ دوباره تلاش کنید.', 4500); } }) .catch(function(){ btn.innerHTML = originalText; btn.disabled = false; btn.classList.remove('opacity-75', 'cursor-not-allowed'); showToast('خطا در ارتباط با سرور. لطفاً اینترنت خود را بررسی کنید.'); }); }
