indicator.style.right = (trackRect.right - btnRect.right) + 'px'; } function setDelivery(orderId, type, btn){ var btnPickup = document.getElementById('btn-pickup-' + orderId); var btnCourier = document.getElementById('btn-courier-' + orderId); if (type === 'pickup'){ secPickup.classList.remove('hidden'); secCourier.classList.add('hidden'); btnPickup.classList.add('active'); } else { secPickup.classList.add('hidden'); secCourier.classList.remove('hidden'); if (!box || typeof L === 'undefined') return; var point = box.getAttribute('data-shop-map').split(',').map(parseFloat); box.innerHTML = ''; var map = L.map(box, { gestureHandling: true, gestureHandlingOptions: { text: { touch: 'برای جابه‌جا کردن نقشه از دو انگشت استفاده کنید', scroll: 'برای بزرگ‌نمایی، Ctrl را نگه دارید و اسکرول کنید', scrollMac: 'برای بزرگ‌نمایی، ⌘ را نگه دارید و اسکرول کنید' } } }).setView(point, 16); } function initCourierMap(orderId){ if (courierMaps[orderId]){ setTimeout(function(){ courierMaps[orderId].invalidateSize(); }, 50); return; } setTimeout(function(){ var mapEl = document.getElementById('interactive-map-' + orderId); L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png', { attribution: '© OpenStreetMap contributors', maxZoom: 19, referrerPolicy: 'strict-origin-when-cross-origin' }).addTo(map); function setCourierPoint(latlng){ placeCourierMarker(orderId, latlng); setAddrMode(orderId, 'new'); lookupAddress(orderId, latlng); } map.on('click', function(e){ setCourierPoint(e.latlng); }); map.on('locationfound', function(e){ map.setView(e.latlng, Math.max(map.getZoom(), 17)); setCourierPoint(e.latlng); }); courierMaps[orderId] = map; map.invalidateSize(); if (SAVED_ADDRESS){ useSavedAddress(orderId); } else { setTimeout(function(){ map.locate({setView:false, enableHighAccuracy:true, maxZoom:17}); }, 150); } }, 50); } function editCourierAddress(orderId){ var box = document.getElementById('courier-edit-' + orderId); var btn = document.getElementById('courier-edit-btn-' + orderId); if (!box) return;
